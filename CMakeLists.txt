cmake_minimum_required(VERSION 4.2)
project(Gateway.Grpc VERSION 0.1.0 LANGUAGES CXX)
include(cmake/Modules.cmake)
include(cmake/Common.cmake)
nc_dependency(Gateway.Api)
nc_dependency(Gateway.Grpc.Api)
add_library(Gateway.Grpc STATIC src/NavalCommand/Gateway/Grpc/Server.cpp)
target_include_directories(Gateway.Grpc PUBLIC public)
target_link_libraries(Gateway.Grpc PUBLIC Gateway.Api Gateway.Grpc.Api spdlog::spdlog)
if(PROJECT_IS_TOP_LEVEL)
 enable_testing()
 add_executable(mock-host app/MockHost.cpp)
 target_include_directories(mock-host PRIVATE mock)
 target_link_libraries(mock-host PRIVATE Gateway.Grpc)
 add_executable(gateway-tests test/ServerTest.cpp)
 target_include_directories(gateway-tests PRIVATE mock)
 target_link_libraries(gateway-tests PRIVATE Gateway.Grpc GTest::gtest_main)
 add_test(NAME gateway COMMAND gateway-tests)
endif()
